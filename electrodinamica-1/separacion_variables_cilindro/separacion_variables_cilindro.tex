\encabezado{Potencial y campo electrostático por separación de variables en un cilindro}
{Electrodinámica I}{P.S., J.A., J.R.}

\problema{1}{Considere un cilindro circular largo (aproximado como infinitamente largo) de radio $R$ centrado en el eje $z$. No hay cargas libres en el espacio ($q = 0$), por lo que el potencial eléctrico $V(s, \phi)$ satisface la ecuación de Laplace en dos dimensiones (asumiendo invarianza a lo largo de $z$):
\begin{equation*}
\nabla^2 V = \frac{1}{s} \frac{\partial}{\partial s}\left(s \frac{\partial V}{\partial s}\right) + \frac{1}{s^2} \frac{\partial^2 V}{\partial \phi^2} = 0
\end{equation*}
Suponga que el potencial sobre la superficie del cilindro ($s = R$) está fijado mediante la siguiente distribución angular:
\begin{equation}\label{eq:borde}
V(R, \phi) = V_0 \sin(2\phi)
\end{equation}
donde $V_0$ es una constante con unidades de potencial eléctrico.}

Se pide:
\begin{enumerate}
    \item \textbf{Separación de Variables:} Plantee la solución general en forma de producto $V(s, \phi) = S(s)\Phi(\phi)$ y obtenga las ecuaciones diferenciales ordinarias para $S(s)$ y $\Phi(\phi)$.
    \item \textbf{Solución Interior ($s < R$):} Halle el potencial eléctrico $V_{\text{int}}(s, \phi)$ en el interior del cilindro, aplicando la condición física de finitud en el origen ($s = 0$).
    \item \textbf{Solución Exterior ($s > R$):} Halle el potencial eléctrico $V_{\text{ext}}(s, \phi)$ en el exterior del cilindro, aplicando la condición de que el potencial decaiga adecuadamente cuando $s \to \infty$.
    \item \textbf{Campo Eléctrico y Densidad de Carga:} Calcule el campo eléctrico $\mathbf{E}(s, \phi)$ en ambas regiones ($s < R$ y $s > R$) utilizando la relación $\mathbf{E} = -\nabla V$, y determine la densidad superficial de carga $\sigma(\phi)$ distribuida sobre la superficie del cilindro ($s = R$).
\end{enumerate}

\emph{Ampliación de la solución:} antes de usar la invarianza axial asumida
en el enunciado, separaremos las tres coordenadas y comprobaremos que las
condiciones de borde la justifican, siempre que el potencial permanezca
acotado cuando $z\to\pm\infty$. La superficie se interpreta como una
interfaz con carga, no como un único conductor equipotencial.

\textbf{Solución:}

\textbf{1. Separación de variables.}\par
Proponemos una solución de la forma factorizada:
\begin{equation*}
V(s, \phi) = S(s) \Phi(\phi)
\end{equation*}
Sustituyendo esta expresión en la ecuación de Laplace en coordenadas cilíndricas:
\begin{equation*}
\frac{1}{s} \frac{d}{ds}\left(s \frac{dS}{ds}\right) \Phi + \frac{1}{s^2} S \frac{d^2\Phi}{d\phi^2} = 0
\end{equation*}
Multiplicamos toda la ecuación por $\frac{s^2}{S(s)\Phi(\phi)}$ para separar las variables:
\begin{equation*}
\frac{s}{S(s)} \frac{d}{ds}\left(s \frac{dS}{ds}\right) = - \frac{1}{\Phi(\phi)} \frac{d^2\Phi}{d\phi^2}
\end{equation*}
Como el lado izquierdo depende exclusivamente de $s$ y el derecho exclusivamente de $\phi$, ambos lados deben ser iguales a una constante de separación, la cual denotaremos como $k^2$:
\begin{align*}
\frac{1}{\Phi(\phi)} \frac{d^2\Phi}{d\phi^2} &= -k^2 \implies \frac{d^2\Phi}{d\phi^2} + k^2 \Phi = 0 \\
\frac{s}{S(s)} \frac{d}{ds}\left(s \frac{dS}{ds}\right) &= k^2 \implies s \frac{d}{ds}\left(s \frac{dS}{ds}\right) - k^2 S = 0
\end{align*}

\subsection*{Resolución de la Ecuación Angular y Radial}

\paragraph{Parte Angular ($\Phi$):}
La solución general para la ecuación armónica angular es:
\begin{equation*}
\Phi(\phi) = A \cos(k\phi) + B \sin(k\phi)
\end{equation*}
Debido a la naturaleza cilíndrica del sistema, el potencial debe ser
unívoco y periódico con período $2\pi$:
\begin{equation*}
    \Phi(\phi)=\Phi(\phi+2\pi).
\end{equation*}

Para determinar los valores permitidos de la constante de separación $k$,
podemos considerar una solución angular compleja de la forma
\begin{equation*}
    \Phi(\phi)=C e^{ik\phi}.
\end{equation*}
Al imponer la condición de periodicidad,
\begin{align*}
    \Phi(\phi+2\pi)
    &= C e^{ik(\phi+2\pi)} \\
    &= C e^{ik\phi}e^{i2\pi k}
    = \Phi(\phi),
\end{align*}
se obtiene
\begin{equation*}
    e^{i2\pi k}=1.
\end{equation*}
Esta igualdad se satisface únicamente cuando
\begin{equation*}
    2\pi k = 2\pi m,
    \qquad m\in\mathbb{Z},
\end{equation*}
por lo que
\begin{equation*}
    k=m,
    \qquad m\in\mathbb{Z}.
\end{equation*}

Como los valores $m$ y $-m$ generan los mismos modos trigonométricos,
pues
\begin{equation*}
    \cos(-m\phi)=\cos(m\phi),
    \qquad
    \sin(-m\phi)=-\sin(m\phi),
\end{equation*}
basta considerar, sin pérdida de generalidad,
\begin{equation*}
    m=0,1,2,\dots
\end{equation*}

Por lo tanto:
\begin{equation*}
    \Phi_m(\phi)
    =
    A_m\cos(m\phi)+B_m\sin(m\phi).
\end{equation*}
\paragraph{Parte Radial ($S$):}
Para $m \neq 0$, la ecuación radial es de tipo Euler-Cauchy:
\begin{equation}\label{eq:radial}
s^2 \frac{d^2 S}{ds^2} + s \frac{dS}{ds} - m^2 S = 0
\end{equation}
Proponiendo soluciones de la forma $S(s) = s^n$, obtenemos $n = \pm m$. Así, la solución radial general es una combinación lineal de potencias:
\begin{equation*}
S_m(s) = C_m s^m + D_m s^{-m}
\end{equation*}
\emph{Nota: si $m=0$, las soluciones radiales son $S_0(s)=C_0+D_0\ln s$. Si hubiese dependencia en el eje $z$, la ecuación radial análoga a \eqref{eq:radial} conduce a funciones de Bessel.}

\textbf{2. Potencial en la región interior ($s<R$).}\par
Para determinar el potencial en el interior del cilindro $V_{\text{int}}(s, \phi)$, aplicamos la condición de contorno física: el potencial debe ser finito en el origen ($s = 0$). Como $\ln(0)$ y $s^{-m}$ divergen cuando $s \to 0$, debemos descartar los términos con coeficientes $D_m$ y $D_0$ fijándolos a cero ($D_m = 0$).

La solución general interior se escribe como una serie superpuesta:
\begin{equation*}
V_{\text{int}}(s, \phi) = \sum_{m=0}^{\infty} s^m \left( A_m \cos(m\phi) + B_m \sin(m\phi) \right)
\end{equation*}

Aplicamos la condición de frontera \eqref{eq:borde} en la superficie del cilindro ($s = R$):
\begin{equation*}
V_{\text{int}}(R, \phi) = \sum_{m=0}^{\infty} R^m \left( A_m \cos(m\phi) + B_m \sin(m\phi) \right) = V_0 \sin(2\phi)
\end{equation*}

Por comparación directa con los términos de la serie de Fourier del lado derecho, observamos que todos los coeficientes se anulan excepto para $m = 2$:
\begin{equation*}
R^2 B_2 = V_0 \implies B_2 = \frac{V_0}{R^2}
\end{equation*}
Todos los demás coeficientes ($A_m$ y los demás $B_m$) son cero. Por lo tanto, el potencial en el interior es:
\begin{equation}\label{eq:potencial-int}
V_{\text{int}}(s, \phi) = V_0 \left( \frac{s}{R} \right)^2 \sin(2\phi)
\end{equation}
\textbf{3. Potencial en la región exterior ($s>R$).}\par
Para el exterior del cilindro $V_{\text{ext}}(s, \phi)$, aplicamos la condición física de que el potencial debe permanecer finito o decaer a cero cuando nos alejamos a grandes distancias ($s \to \infty$). Por lo tanto, debemos descartar las potencias crecientes $s^m$ fijando $C_m = 0$.

La solución general exterior es:
\begin{equation*}
V_{\text{ext}}(s, \phi) = \sum_{m=0}^{\infty} s^{-m} \left( A'_m \cos(m\phi) + B'_m \sin(m\phi) \right)
\end{equation*}

Aplicando nuevamente la condición de frontera \eqref{eq:borde} en $s = R$:
\begin{equation*}
V_{\text{ext}}(R, \phi) = \sum_{m=0}^{\infty} R^{-m} \left( A'_m \cos(m\phi) + B'_m \sin(m\phi) \right) = V_0 \sin(2\phi)
\end{equation*}

Nuevamente, por ortogonalidad y comparación de coeficientes, el único término no nulo corresponde a $m = 2$:
\begin{equation*}
R^{-2} B'_2 = V_0 \implies B'_2 = V_0 R^2
\end{equation*}

Por consiguiente, el potencial eléctrico en la región exterior es:
\begin{equation}\label{eq:potencial-ext}
V_{\text{ext}}(s, \phi) = V_0 \left( \frac{R}{s} \right)^2 \sin(2\phi)
\end{equation}

\textbf{4. Campo eléctrico y densidad superficial.}\par

Para obtener la densidad superficial de carga sobre el cilindro utilizamos
la condición de contorno para la componente normal del campo eléctrico:
\begin{equation}\label{eq:salto}
    \sigma(\phi)
    =
    \varepsilon_0
    \left[
        E_{\text{ext},s}(R,\phi)
        -
        E_{\text{int},s}(R,\phi)
    \right].
\end{equation}

La normal a la superficie cilíndrica $s=R$ apunta en la dirección radial,
\begin{equation*}
    \hat{\mathbf n}=\hat{\mathbf s},
\end{equation*}
por lo que la componente normal del campo eléctrico es simplemente su
componente radial. Como
\begin{equation*}
    \mathbf E=-\nabla V,
\end{equation*}
tenemos
\begin{equation*}
    E_s
    =
    \mathbf E\cdot\hat{\mathbf s}
    =
    -\frac{\partial V}{\partial s}.
\end{equation*}

Equivalentemente, la derivada normal del potencial sobre la superficie es
\begin{equation*}
    \frac{\partial V}{\partial n}
    =
    \nabla V\cdot\hat{\mathbf n}
    =
    \frac{\partial V}{\partial s},
\end{equation*}
de modo que la Ec.~\eqref{eq:salto} puede escribirse directamente en
terminos de las derivadas radiales de los potenciales interior y exterior.

\medskip

\textbf{Campo radial interior.}\par

A partir del potencial interior \eqref{eq:potencial-int},
\begin{align*}
    E_{\text{int},s}(s,\phi)
    &=
    -\frac{\partial V_{\text{int}}}{\partial s}
    \\
    &=
    -\frac{\partial}{\partial s}
    \left[
        V_0\frac{s^2}{R^2}\sin(2\phi)
    \right]
    \\
    &=
    -\frac{2V_0s}{R^2}\sin(2\phi).
\end{align*}

Evaluando justo en la cara interior de la superficie, $s=R^{-}$,
\begin{equation}\label{eq:eint}
    E_{\text{int},s}(R,\phi)
    =
    -\frac{2V_0}{R}\sin(2\phi).
\end{equation}

\medskip

\textbf{Campo radial exterior.}\par

A partir del potencial exterior \eqref{eq:potencial-ext},
\begin{align*}
    E_{\text{ext},s}(s,\phi)
    &=
    -\frac{\partial V_{\text{ext}}}{\partial s}
    \\
    &=
    -\frac{\partial}{\partial s}
    \left[
        V_0R^2s^{-2}\sin(2\phi)
    \right]
    \\
    &=
    \frac{2V_0R^2}{s^3}\sin(2\phi).
\end{align*}

Evaluando justo en la cara exterior de la superficie, $s=R^{+}$,
\begin{equation}\label{eq:eext}
    E_{\text{ext},s}(R,\phi)
    =
    \frac{2V_0}{R}\sin(2\phi).
\end{equation}

Finalmente, sustituyendo \eqref{eq:eint} y \eqref{eq:eext} en la condición
de salto \eqref{eq:salto},
\begin{align*}
    \sigma(\phi)
    &=
    \varepsilon_0
    \left[
        \frac{2V_0}{R}\sin(2\phi)
        -
        \left(
            -\frac{2V_0}{R}\sin(2\phi)
        \right)
    \right]
    \\
    &=
    \frac{4\varepsilon_0V_0}{R}\sin(2\phi).
\end{align*}

Por consiguiente,
\begin{equation}\label{eq:sigma}
    \boxed{
        \sigma(\phi)
        =
        \frac{4\varepsilon_0V_0}{R}\sin(2\phi)
    }.
\end{equation}

Este resultado muestra que para sostener esta distribución de potencial armónico en la superficie del cilindro, se requiere una densidad superficial de carga no uniforme que oscila azimutalmente con la misma periodicidad $\sin(2\phi)$.
